⚡ Tool Migrasi Data ke Supabase

Alat ini akan mengunggah seluruh katalog game PC & PS2 kamu dari file lokal ke database Supabase (PostgreSQL) secara otomatis.

CREATE TABLE IF NOT EXISTS public.games ( id TEXT PRIMARY KEY, title TEXT NOT NULL, category TEXT NOT NULL, size_gb NUMERIC, platform TEXT, cover TEXT, banner_url TEXT, game_info JSONB, download_links JSONB, custom BOOLEAN DEFAULT true, created_at TIMESTAMPTZ DEFAULT NOW() ); ALTER TABLE public.games ENABLE ROW LEVEL SECURITY; DROP POLICY IF EXISTS "Allow public read" ON public.games; DROP POLICY IF EXISTS "Allow public write" ON public.games; DROP POLICY IF EXISTS "Allow public write for migration" ON public.games; CREATE POLICY "Allow public read" ON public.games FOR SELECT USING (true); CREATE POLICY "Allow public write" ON public.games FOR ALL USING (true) WITH CHECK (true);
